    <script>
        // Sample Mod Data
        let mods = [
            { id: 1, name: "VIP Injector Panel", price: "Rs. 500 / Month" },
            { id: 2, name: "Aimbot + ESP Mod", price: "Rs. 800 / Month" },
            { id: 3, name: "Ultra Headshot Config", price: "Rs. 300 / Lifetime" }
        ];

        // Handle Login Authentication
        function handleLogin() {
            const user = document.getElementById('username').value;
            const pass = document.getElementById('password').value;

            if (user === "" || pass === "") {
                alert("कृपया Username र Password दुबै राख्नुहोस्!");
                return;
            }

            // Authentication logic
            let role = "User";
            if (user === "admin" && pass === "admin123") {
                role = "Admin";
            }

            document.getElementById('login-box').style.display = 'none';
            document.getElementById('dashboard').style.display = 'block';
            document.getElementById('user-info').innerHTML = `
                लगइन भएको छ: <b>${user}</b> (${role}) 
                <button class="btn btn-danger" style="margin-left: 10px;" onclick="location.reload()">Logout</button>
            `;

            // Show Admin Panel if logged in as admin
            if (role === "Admin") {
                document.getElementById('admin-panel').style.display = 'block';
            }

            renderMods();
        }

        // Display Mods in UI
        function renderMods() {
            const container = document.getElementById('mod-list');
            container.innerHTML = "";

            mods.forEach(mod => {
                const card = document.createElement('div');
                card.className = 'card';
                card.innerHTML = `
                    <h3>${mod.name}</h3>
                    <p class="price">${mod.price}</p>
                    <button class="btn btn-buy" onclick="buyMod('${mod.name}')">Buy Now</button>
                `;
                container.appendChild(card);
            });
        }

        // Buy Action
        function buyMod(modName) {
            alert(`तपाईंले '${modName}' खरिद गर्न रोज्नुभएको छ। Payment प्रोसेसको लागि Admin सँग सम्पर्क गर्नुहोस्।`);
        }

        // Admin Action: Add New Mod
        function addNewMod() {
            const name = document.getElementById('new-mod-name').value;
            const price = document.getElementById('new-mod-price').value;

            if (name === "" || price === "") {
                alert("मोडको नाम र मूल्य अनिवार्य छ!");
                return;
            }

            mods.push({ id: Date.now(), name: name, price: price });
            renderMods();

            document.getElementById('new-mod-name').value = "";
            document.getElementById('new-mod-price').value = "";
            alert("नयाँ मोड सफलतापूर्वक थपियो!");
        }
    </script>
